词法分析 - go/scanner

go/scanner 包提供了 Go 源代码的词法分析功能，将源代码分解为 Token 序列。这是构建编译器、代码分析工具的基础。

基础用法

扫描代码

📝 基本扫描示例

package main

import (
    "fmt"
    "go/scanner"
    "go/token"
)

func main() {
    // 创建文件集
    fset := token.NewFileSet()

    // 源代码
    src := []byte(`package main

import "fmt"

func main() {
    fmt.Println("Hello, World!")
}`)

    // 创建文件
    file := fset.AddFile("", fset.Base(), len(src))

    // 创建扫描器
    var s scanner.Scanner
    s.Init(file, src, nil, scanner.ScanComments)

    // 扫描 Token
    for {
        pos, tok, lit := s.Scan()
        if tok == token.EOF {
            break
        }
        fmt.Printf("%s\t%s\t%q\n",
            fset.Position(pos),
            tok,
            lit,
        )
    }
}

Token 类型

📖 Token 分类

package main

import (
    "fmt"
    "go/token"
)

func main() {
    // 特殊 Token
    fmt.Println(token.ILLEGAL)  // 非法 Token
    fmt.Println(token.EOF)      // 文件结束

    // 标识符和字面量
    fmt.Println(token.IDENT)    // 标识符
    fmt.Println(token.INT)      // 整数
    fmt.Println(token.FLOAT)    // 浮点数
    fmt.Println(token.IMAG)     // 虚数
    fmt.Println(token.CHAR)     // 字符
    fmt.Println(token.STRING)   // 字符串
    fmt.Println(token.COMMENT)  // 注释

    // 运算符
    fmt.Println(token.ADD)      // +
    fmt.Println(token.SUB)      // -
    fmt.Println(token.MUL)      // *
    fmt.Println(token.QUO)      // /
    fmt.Println(token.REM)      // %

    // 关键字
    fmt.Println(token.BREAK)    // break
    fmt.Println(token.CASE)     // case
    fmt.Println(token.FUNC)     // func
    fmt.Println(token.IF)       // if
    fmt.Println(token.RETURN)   // return
}

扫描模式

📝 Mode 配置

package main

import (
    "fmt"
    "go/scanner"
    "go/token"
)

func scanWithMode(mode scanner.Mode) {
    fset := token.NewFileSet()
    src := []byte(`// 注释
package main /* 多行注释 */`)
    
    file := fset.AddFile("", fset.Base(), len(src))
    
    var s scanner.Scanner
    s.Init(file, src, nil, mode)
    
    fmt.Printf("Mode: %v\n", mode)
    for {
        pos, tok, lit := s.Scan()
        if tok == token.EOF {
            break
        }
        if tok == token.COMMENT {
            fmt.Printf("  Comment: %s\n", lit)
        }
    }
}

func main() {
    // 扫描注释
    scanWithMode(scanner.ScanComments)
    
    // 不扫描注释 (默认)
    scanWithMode(0)
    
    // 允许非法字面量
    scanWithMode(scanner.AllowIllegalChars)
}

错误处理

📝 自定义错误处理器

package main

import (
    "fmt"
    "go/scanner"
    "go/token"
)

func main() {
    fset := token.NewFileSet()
    
    // 包含错误的源代码
    src := []byte(`package main

func invalid( {
    // 缺少右括号
}`)
    
    file := fset.AddFile("", fset.Base(), len(src))
    
    // 错误计数器
    errorCount := 0
    
    // 自定义错误处理函数
    errorHandler := func(pos token.Position, msg string) {
        errorCount++
        fmt.Printf("Error %d at %s: %s\n",
            errorCount, pos, msg)
    }
    
    var s scanner.Scanner
    s.Init(file, src, errorHandler, scanner.ScanComments)
    
    for {
        pos, tok, lit := s.Scan()
        if tok == token.EOF {
            break
        }
        if tok == token.ILLEGAL {
            fmt.Printf("Illegal token at %s: %q\n",
                fset.Position(pos), lit)
        }
    }
    
    fmt.Printf("Total errors: %d\n", errorCount)
}

实用工具

Fprint 打印 Token

package main

import (
    "os"
    "go/scanner"
    "go/token"
)

func main() {
    fset := token.NewFileSet()
    src := []byte(`package main; import "fmt"; func main() { fmt.Println("Hi") }`)
    
    file := fset.AddFile("", fset.Base(), len(src))
    
    var s scanner.Scanner
    s.Init(file, src, nil, scanner.ScanComments)
    
    // 使用 Fprint 打印 Token
    scanner.Fprint(os.Stdout, fset, s.TokenText())
}

TokenText 获取文本

var s scanner.Scanner
s.Init(file, src, nil, scanner.ScanComments)

for {
    pos, tok, _ := s.Scan()
    if tok == token.EOF {
        break
    }
    
    // 获取 Token 的原始文本
    text := s.TokenText()
    fmt.Printf("%s: %s\n", tok, text)
}

应用场景

📝 统计代码元素

package main

import (
    "fmt"
    "go/scanner"
    "go/token"
    "os"
)

type CodeStats struct {
    Lines     int
    Comments  int
    Functions int
    Imports   int
}

func AnalyzeFile(filename string) (*CodeStats, error) {
    src, err := os.ReadFile(filename)
    if err != nil {
        return nil, err
    }
    
    fset := token.NewFileSet()
    file := fset.AddFile(filename, fset.Base(), len(src))
    
    stats := &CodeStats{}
    var s scanner.Scanner
    s.Init(file, src, nil, scanner.ScanComments)
    
    for {
        _, tok, lit := s.Scan()
        switch tok {
        case token.EOF:
            return stats, nil
        case token.COMMENT:
            stats.Comments++
        case token.FUNC:
            stats.Functions++
        case token.IMPORT:
            stats.Imports++
        case token.STRING:
            // 统计字符串字面量
        }
    }
}

func main() {
    stats, err := AnalyzeFile("main.go")
    if err != nil {
        fmt.Printf("Error: %v\n", err)
        return
    }
    
    fmt.Printf("Functions: %d\n", stats.Functions)
    fmt.Printf("Imports: %d\n", stats.Imports)
    fmt.Printf("Comments: %d\n", stats.Comments)
}

✅ 使用建议

  • 使用 FileSet: 始终使用 token.FileSet 管理位置信息
  • 错误处理: 提供自定义错误处理函数
  • 扫描模式: 根据需要选择是否扫描注释
  • TokenText: 使用 TokenText() 获取原始文本
  • EOF 检查: 始终检查 token.EOF 结束扫描
