语法树 - go/ast

go/ast 包提供了 Go 抽象语法树 (AST) 的数据结构和操作函数。AST 是编译器对源代码的结构化表示，用于代码分析、转换和生成。

基础用法

解析文件

📝 解析 Go 文件

package main

import (
    "fmt"
    "go/ast"
    "go/parser"
    "go/token"
)

func main() {
    // 源代码
    src := `package main

import "fmt"

func main() {
    fmt.Println("Hello")
}`

    // 创建文件集
    fset := token.NewFileSet()

    // 解析源代码
    file, err := parser.ParseFile(fset, "", src, 0)
    if err != nil {
        panic(err)
    }

    // 打印 AST
    ast.Fprint(nil, fset, file, nil)
}

遍历 AST

使用 Inspect

📝 遍历所有节点

package main

import (
    "fmt"
    "go/ast"
    "go/parser"
    "go/token"
)

func main() {
    src := `package main

import "fmt"

func main() {
    fmt.Println("Hello")
}`

    fset := token.NewFileSet()
    file, _ := parser.ParseFile(fset, "", src, 0)

    // 使用 Inspect 遍历
    ast.Inspect(file, func(n ast.Node) bool {
        if n != nil {
            fmt.Printf("%T\n", n)
        }
        return true // 继续遍历子节点
    })
}

查找特定节点

📝 查找函数声明

package main

import (
    "fmt"
    "go/ast"
    "go/parser"
    "go/token"
)

func main() {
    src := `package main

func add(a, b int) int {
    return a + b
}

func main() {
    result := add(1, 2)
    fmt.Println(result)
}`

    fset := token.NewFileSet()
    file, _ := parser.ParseFile(fset, "", src, 0)

    // 查找所有函数
    ast.Inspect(file, func(n ast.Node) bool {
        if fn, ok := n.(*ast.FuncDecl); ok {
            fmt.Printf("Function: %s\n", fn.Name.Name)
            
            // 打印参数
            if fn.Type.Params != nil {
                fmt.Print("  Parameters: ")
                for i, param := range fn.Type.Params.List {
                    if i > 0 {
                        fmt.Print(", ")
                    }
                    for j, name := range param.Names {
                        if j > 0 {
                            fmt.Print(", ")
                        }
                        fmt.Print(name.Name)
                    }
                }
                fmt.Println()
            }
            
            // 打印返回值
            if fn.Type.Results != nil && len(fn.Type.Results.List) > 0 {
                resultType := fn.Type.Results.List[0].Type
                fmt.Printf("  Returns: %T\n", resultType)
            }
        }
        return true
    })
}

实用示例

统计函数数量

📝 代码统计工具

package main

import (
    "fmt"
    "go/ast"
    "go/parser"
    "go/token"
)

type CodeStats struct {
    Functions   int
    Methods     int
    Imports     int
    Structs     int
    Interfaces  int
}

func AnalyzeFile(filename string) (*CodeStats, error) {
    fset := token.NewFileSet()
    file, err := parser.ParseFile(fset, filename, nil, 0)
    if err != nil {
        return nil, err
    }
    
    stats := &CodeStats{}
    
    ast.Inspect(file, func(n ast.Node) bool {
        switch x := n.(type) {
        case *ast.FuncDecl:
            if x.Recv != nil {
                stats.Methods++
            } else {
                stats.Functions++
            }
        case *ast.GenDecl:
            if x.Tok == token.IMPORT {
                stats.Imports += len(x.Specs)
            }
            for _, spec := range x.Specs {
                switch spec.(type) {
                case *ast.TypeSpec:
                    ts := spec.(*ast.TypeSpec)
                    switch ts.Type.(type) {
                    case *ast.StructType:
                        stats.Structs++
                    case *ast.InterfaceType:
                        stats.Interfaces++
                    }
                }
            }
        }
        return true
    })
    
    return stats, nil
}

func main() {
    stats, err := AnalyzeFile("main.go")
    if err != nil {
        panic(err)
    }
    
    fmt.Printf("Functions: %d\n", stats.Functions)
    fmt.Printf("Methods: %d\n", stats.Methods)
    fmt.Printf("Imports: %d\n", stats.Imports)
    fmt.Printf("Structs: %d\n", stats.Structs)
    fmt.Printf("Interfaces: %d\n", stats.Interfaces)
}

查找所有调用

📝 函数调用分析

package main

import (
    "fmt"
    "go/ast"
    "go/parser"
    "go/token"
)

func FindCallExprs(filename string) {
    fset := token.NewFileSet()
    file, _ := parser.ParseFile(fset, filename, nil, 0)
    
    ast.Inspect(file, func(n ast.Node) bool {
        if call, ok := n.(*ast.CallExpr); ok {
            // 获取被调用的函数名
            switch fun := call.Fun.(type) {
            case *ast.Ident:
                fmt.Printf("Call: %s()\n", fun.Name)
            case *ast.SelectorExpr:
                if ident, ok := fun.X.(*ast.Ident); ok {
                    fmt.Printf("Call: %s.%s()\n",
                        ident.Name, fun.Sel.Name)
                }
            }
            
            // 打印参数数量
            fmt.Printf("  Arguments: %d\n", len(call.Args))
        }
        return true
    })
}

func main() {
    FindCallExprs("main.go")
}

生成代码

📝 AST 代码生成

package main

import (
    "bytes"
    "fmt"
    "go/ast"
    "go/format"
    "go/token"
)

func GenerateHello() []byte {
    // 创建文件
    file := &ast.File{
        Name: ast.NewIdent("main"),
    }
    
    // 创建 main 函数
    mainFunc := &ast.FuncDecl{
        Name: ast.NewIdent("main"),
        Type: &ast.FuncType{
            Params: &ast.FieldList{},
        },
        Body: &ast.BlockStmt{
            List: []ast.Stmt{
                &ast.ExprStmt{
                    X: &ast.CallExpr{
                        Fun: &ast.SelectorExpr{
                            X:  ast.NewIdent("fmt"),
                            Sel: ast.NewIdent("Println"),
                        },
                        Args: []ast.Expr{
                            &ast.BasicLit{
                                Kind:  token.STRING,
                                Value: `"Hello, World!"`,
                            },
                        },
                    },
                },
            },
        },
    }
    
    file.Decls = []ast.Decl{mainFunc}
    
    // 格式化输出
    var buf bytes.Buffer
    fset := token.NewFileSet()
    format.Node(&buf, fset, file)
    
    return buf.Bytes()
}

func main() {
    code := GenerateHello()
    fmt.Print(string(code))
}

节点类型详解

声明节点 (Declarations)

📖 FuncDecl - 函数声明

type FuncDecl struct {
    Doc  *CommentGroup  // 关联的文档注释
    Recv *FieldList     // 接收者 (方法); 无则为 nil
    Name *Ident         // 函数名
    Type *FuncType      // 函数类型 (参数、结果)
    Body *BlockStmt     // 函数体; 无则为 nil
}

// 示例代码:
func (p *Person) Say(msg string) string {
    return msg
}

// AST 结构:
*ast.FuncDecl {
    .Recv: *ast.FieldList {
        .List: []*ast.Field {
            {
                .Names: []*ast.Ident {{.Name: "p"}},
                .Type:  &ast.StarExpr{X: &ast.Ident{Name: "Person"}},
            },
        },
    },
    .Name: *ast.Ident {.Name: "Say"},
    .Type: *ast.FuncType {
        .Params: *ast.FieldList {...},
        .Results: *ast.FieldList {...},
    },
    .Body: *ast.BlockStmt {...},
}

📖 GenDecl - 通用声明

type GenDecl struct {
    Doc  *CommentGroup  // 关联的文档注释
    TokPos token.Pos   // import/var/const/type 的位置
    Tok  token.Token    // IMPORT, VAR, CONST, 或 TYPE
    Lparen token.Pos   // 左括号位置
    Specs []Spec        // 声明列表
    Rparen token.Pos   // 右括号位置
}

// 示例代码:
import (
    "fmt"
    "os"
)

var (
    count int
    name  string
)

const Pi = 3.14

type Person struct {
    Name string
}

// Tok 字段值:
// token.IMPORT - import 声明
// token.VAR    - var 声明
// token.CONST  - const 声明
// token.TYPE   - type 声明

📖 ValueSpec - 值声明 (var/const)

type ValueSpec struct {
    Doc     *CommentGroup  // 关联的文档注释
    Names   []*Ident       // 变量/常量名列表
    Type    Expr           // 类型表达式; 无则为 nil
    Values  []Expr         // 初始值; 无则为 nil
    Comment *CommentGroup  // 行尾注释
}

// 示例代码:
var count int = 0
const MaxSize = 100
var x, y, z int

// AST 结构 (var count int = 0):
*ast.ValueSpec {
    .Names: []*ast.Ident {{.Name: "count"}},
    .Type:  &ast.Ident{Name: "int"},
    .Values: []ast.Expr {
        &ast.BasicLit{Kind: token.INT, Value: "0"},
    },
}

📖 TypeSpec - 类型声明

type TypeSpec struct {
    Doc     *CommentGroup  // 关联的文档注释
    Name    *Ident         // 类型名
    TypeParams *FieldList  // 泛型参数; Go 1.18+
    Assign  token.Pos      // "= "的位置; 无则为 0
    Type    Expr           // 类型表达式
    Comment *CommentGroup  // 行尾注释
}

// 示例代码:
type Person struct {
    Name string
}

type Handler func(string) error

type IntList []int

// AST 结构 (type Person struct{...}):
*ast.TypeSpec {
    .Name: &ast.Ident{Name: "Person"},
    .Type: &ast.StructType{
        .Fields: &ast.FieldList{
            .List: []*ast.Field{
                {
                    .Names: []*ast.Ident{{.Name: "Name"}},
                    .Type:  &ast.Ident{Name: "string"},
                },
            },
        },
    },
}

语句节点 (Statements)

📖 BlockStmt - 代码块

type BlockStmt struct {
    Lbrace token.Pos   // 左括号 { 的位置
    List   []Stmt      // 语句列表
    Rbrace token.Pos   // 右括号 } 的位置
}

// 示例代码:
{
    x := 1
    fmt.Println(x)
}

📖 IfStmt - if 语句

type IfStmt struct {
    If   token.Pos    // if 关键字位置
    Init Stmt         // 初始化语句; 无则为 nil
    Cond Expr         // 条件表达式
    Body *BlockStmt   // if 分支代码块
    Else Stmt         // else 分支; 无则为 nil
}

// 示例代码:
if x > 0 {
    fmt.Println("positive")
} else {
    fmt.Println("non-positive")
}

// 带初始化:
if err := doSomething(); err != nil {
    return err
}

📖 ForStmt - for 循环

type ForStmt struct {
    For  token.Pos    // for 关键字位置
    Init Stmt         // 初始化语句; 无则为 nil
    Cond Expr         // 条件表达式; 无则为 nil
    Post Stmt         // 后置语句; 无则为 nil
    Body *BlockStmt   // 循环体
}

// 示例代码:
// 标准 for 循环
for i := 0; i < 10; i++ {
    fmt.Println(i)
}

// while 风格
for condition {
    // ...
}

// 无限循环
for {
    // ...
}

📖 RangeStmt - range 循环

type RangeStmt struct {
    For        token.Pos   // for 关键字位置
    Key, Value Expr        // key, value 变量
    TokPos     token.Pos   // = 或 := 的位置
    Tok        token.Token // ASSIGN 或 DEFINE
    Body       *BlockStmt  // 循环体
    X          Expr        // 要 range 的表达式
}

// 示例代码:
for i, v := range slice {
    fmt.Printf("%d: %v\n", i, v)
}

for key, value := range m {
    fmt.Printf("%s: %v\n", key, value)
}

for _, ch := range "hello" {
    fmt.Printf("%c\n", ch)
}

表达式节点 (Expressions)

📖 Ident - 标识符

type Ident struct {
    NamePos token.Pos  // 标识符位置
    Name    string     // 标识符名称
}

// 示例代码:
x
Person
fmt.Println

// AST 结构:
*ast.Ident {.Name: "x"}
*ast.Ident {.Name: "Person"}
*ast.Ident {.Name: "Println"}

📖 BasicLit - 字面量

type BasicLit struct {
    ValuePos token.Pos    // 字面量位置
    Kind     token.Token  // INT, FLOAT, IMAG, CHAR, STRING
    Value    string       // 字面量值
}

// 示例代码:
42
3.14
"hello"
'a'
1e10

// AST 结构:
*ast.BasicLit {
    .Kind: token.INT,
    .Value: "42",
}
*ast.BasicLit {
    .Kind: token.STRING,
    .Value: "\"hello\"",
}

📖 CallExpr - 函数调用

type CallExpr struct {
    Fun      Expr        // 函数表达式
    Lparen   token.Pos   // 左括号位置
    Args     []Expr      // 参数列表
    Ellipsis token.Pos   // ... 的位置; 无则为 0
    Rparen   token.Pos   // 右括号位置
}

// 示例代码:
fmt.Println("hello", 42)
make([]int, 10)
append(slice, elems...)

// AST 结构 (fmt.Println("hello")):
*ast.CallExpr {
    .Fun: &ast.SelectorExpr{
        .X:   &ast.Ident{Name: "fmt"},
        .Sel: &ast.Ident{Name: "Println"},
    },
    .Args: []ast.Expr{
        &ast.BasicLit{Kind: token.STRING, Value: "\"hello\""},
    },
}

📖 SelectorExpr - 选择器表达式

type SelectorExpr struct {
    X   Expr   // 表达式
    Sel *Ident // 选择的字段/方法名
}

// 示例代码:
fmt.Println
person.Name
response.StatusCode

// AST 结构 (fmt.Println):
*ast.SelectorExpr {
    .X:   &ast.Ident{Name: "fmt"},
    .Sel: &ast.Ident{Name: "Println"},
}

📖 BinaryExpr - 二元表达式

type BinaryExpr struct {
    X     Expr          // 左操作数
    Op    token.Token   // 运算符
    OpPos token.Pos     // 运算符位置
    Y     Expr          // 右操作数
}

// 示例代码:
a + b
x == 10
name != ""
count >= 0

// AST 结构 (a + b):
*ast.BinaryExpr {
    .X:   &ast.Ident{Name: "a"},
    .Op:  token.ADD,
    .Y:   &ast.Ident{Name: "b"},
}

类型节点 (Types)

📖 StructType - 结构体类型

type StructType struct {
    Struct     token.Pos   // struct 关键字位置
    Fields     *FieldList  // 字段列表
    Incomplete bool        // 如果有未解析的字段则为 true
}

// 示例代码:
type Person struct {
    Name string
    Age  int
}

// AST 结构:
*ast.StructType {
    .Fields: &ast.FieldList{
        .List: []*ast.Field{
            {
                .Names: []*ast.Ident{{.Name: "Name"}},
                .Type:  &ast.Ident{Name: "string"},
            },
            {
                .Names: []*ast.Ident{{.Name: "Age"}},
                .Type:  &ast.Ident{Name: "int"},
            },
        },
    },
}

📖 InterfaceType - 接口类型

type InterfaceType struct {
    Interface  token.Pos   // interface 关键字位置
    Methods    *FieldList  // 方法列表
    Incomplete bool        // 如果有未解析的方法则为 true
}

// 示例代码:
type Reader interface {
    Read(p []byte) (n int, err error)
}

type Writer interface {
    Write(p []byte) (n int, err error)
}

📖 ArrayType - 数组/切片类型

type ArrayType struct {
    Lbrack token.Pos  // 左括号位置
    Len    Expr       // 数组长度; nil 表示切片
    Elt    Expr       // 元素类型
}

// 示例代码:
[]int           // 切片: Len = nil
[10]int         // 数组：Len = &ast.BasicLit{Value: "10"}
[...]byte        // 数组：Len = &ast.Ellipsis
[][]string      // 切片的切片

// AST 结构 ([]int):
*ast.ArrayType {
    .Len:  nil,
    .Elt:  &ast.Ident{Name: "int"},
}

// AST 结构 ([10]int):
*ast.ArrayType {
    .Len:  &ast.BasicLit{Kind: token.INT, Value: "10"},
    .Elt:  &ast.Ident{Name: "int"},
}

📖 MapType - Map 类型

type MapType struct {
    Map   token.Pos  // map 关键字位置
    Key   Expr       // 键类型
    Value Expr       // 值类型
}

// 示例代码:
map[string]int
map[int]*Person
map[string][]byte

// AST 结构 (map[string]int):
*ast.MapType {
    .Key:   &ast.Ident{Name: "string"},
    .Value: &ast.Ident{Name: "int"},
}

📖 ChanType - Channel 类型

type ChanType struct {
    Begin token.Pos      // chan 关键字位置或 <- 位置
    Dir   ChanDir        // 方向：SEND, RECV, SEND|RECV
    Value Expr           // 元素类型
}

type ChanDir int

const (
    SEND ChanDir = 1 // <- chan
    RECV ChanDir = 2 // chan <-
)

// 示例代码:
chan int            // 双向 channel
chan<- int          // 只接收
chan<- string       // 只发送

// AST 结构 (chan int):
*ast.ChanType {
    .Dir:   ast.SEND | ast.RECV,
    .Value: &ast.Ident{Name: "int"},
}

// AST 结构 (chan<- string):
*ast.ChanType {
    .Dir:   ast.SEND,
    .Value: &ast.Ident{Name: "string"},
}

最佳实践

✅ 使用建议

  • 使用 FileSet: 始终使用 token.FileSet 管理位置
  • 类型断言: 使用 type switch 处理不同节点
  • Inspect 返回: 返回 false 可跳过子节点遍历
  • 格式化输出: 使用 format.Node 格式化 AST
  • 错误处理: 解析时始终检查错误
