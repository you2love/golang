Embed 简介

embed 是 Go 1.16 引入的标准库，用于在编译时将文件嵌入到 Go 程序中。它提供了一种简单、类型安全的方式来访问嵌入的文件，无需在运行时读取文件系统。

Embed 的核心特性

  • 编译时嵌入：文件在编译时被嵌入到二进制文件中
  • 类型安全：提供 string、[]byte 和 fs.FS 类型
  • 跨平台：自动处理不同操作系统的路径分隔符
  • 零依赖：不需要额外的依赖管理
  • 简单易用：通过 //go:embed 指令声明

嵌入单个文件为字符串

package main

import (
    _ "embed"
    "fmt"
)

//go:embed config.txt
var configFile string

func main() {
    fmt.Println("配置文件内容:")
    fmt.Println(configFile)
}

config.txt 文件内容：

app.name=MyApp
app.version=1.0.0
debug=true

嵌入单个文件为字节切片

package main

import (
    _ "embed"
    "fmt"
)

//go:embed logo.png
var logo []byte

func main() {
    fmt.Printf("Logo 大小: %d 字节\n", len(logo))
    fmt.Printf("Logo 前几个字节: %x\n", logo[:min(10, len(logo))])
}

嵌入多个文件为文件系统

package main

import (
    "embed"
    "fmt"
    "io/fs"
)

//go:embed templates/*.html
var templatesFS embed.FS

func main() {
    fmt.Println("读取模板文件:")
    
    // 遍历文件系统
    err := fs.WalkDir(templatesFS, ".", func(path string, d fs.DirEntry, err error) error {
        if err != nil {
            return err
        }
        
        if !d.IsDir() {
            fmt.Printf("文件: %s\n", path)
            
            // 读取文件内容
            content, err := fs.ReadFile(templatesFS, path)
            if err == nil {
                fmt.Printf("内容: %s\n", string(content))
            }
        }
        return nil
    })
    
    if err != nil {
        fmt.Println("错误:", err)
    }
}

嵌入整个目录

package main

import (
    "embed"
    "fmt"
    "io/fs"
    "log"
    "net/http"
)

//go:embed static
var staticFS embed.FS

func main() {
    // 列出嵌入的静态文件
    err := fs.WalkDir(staticFS, "static", func(path string, d fs.DirEntry, err error) error {
        if err != nil {
            return err
        }
        if d.IsDir() {
            fmt.Printf("目录: %s\n", path)
        } else {
            fmt.Printf("文件: %s\n", path)
        }
        return nil
    })
    
    if err != nil {
        log.Fatal(err)
    }
    
    // 创建 HTTP 服务器
    // 注意：需要使用 fs.Sub 来去掉 "static" 前缀
    staticFiles, err := fs.Sub(staticFS, "static")
    if err != nil {
        log.Fatal(err)
    }
    
    http.Handle("/", http.FileServer(http.FS(staticFiles)))
    fmt.Println("服务器启动在 http://localhost:8080")
    log.Fatal(http.ListenAndServe(":8080", nil))
}

嵌入 HTML 模板

package main

import (
    "embed"
    "fmt"
    "html/template"
)

//go:embed templates/index.html
var indexTemplate string

//go:embed templates/*.html
var templatesFS embed.FS

func main() {
    // 方法1：直接使用字符串
    tmpl, err := template.New("index").Parse(indexTemplate)
    if err != nil {
        fmt.Println("解析模板错误:", err)
        return
    }
    
    // 方法2：使用文件系统
    tmplFS, err := template.ParseFS(templatesFS, "templates/*.html")
    if err != nil {
        fmt.Println("解析模板错误:", err)
        return
    }
    
    data := struct {
        Title   string
        Message string
    }{
        Title:   "欢迎",
        Message: "Hello, Go Embed!",
    }
    
    err = tmpl.Execute(nil, data)
    if err != nil {
        fmt.Println("执行模板错误:", err)
    }
    
    // 使用文件系统模板
    err = tmplFS.ExecuteTemplate(nil, "index.html", data)
    if err != nil {
        fmt.Println("执行模板错误:", err)
    }
}

嵌入配置文件（JSON/YAML）

package main

import (
    _ "embed"
    "encoding/json"
    "fmt"
)

//go:embed config.json
var configFile string

type Config struct {
    AppName string `json:"app_name"`
    Version string `json:"version"`
    Debug   bool   `json:"debug"`
    Server  struct {
        Host string `json:"host"`
        Port int    `json:"port"`
    } `json:"server"`
}

func main() {
    var config Config
    
    err := json.Unmarshal([]byte(configFile), &config)
    if err != nil {
        fmt.Println("解析配置错误:", err)
        return
    }
    
    fmt.Printf("应用名称: %s\n", config.AppName)
    fmt.Printf("版本: %s\n", config.Version)
    fmt.Printf("调试模式: %v\n", config.Debug)
    fmt.Printf("服务器: %s:%d\n", config.Server.Host, config.Server.Port)
}

config.json 文件内容：

{
    "app_name": "MyGoApp",
    "version": "1.0.0",
    "debug": true,
    "server": {
        "host": "localhost",
        "port": 8080
    }
}

使用通配符

package main

import (
    "embed"
    "fmt"
    "io/fs"
)

// 嵌入所有 .txt 文件
//go:embed *.txt
var textFiles embed.FS

// 嵌入所有文件（包括子目录）
//go:embed all:*
var allFiles embed.FS

// 嵌入特定模式
//go:embed data/*.json
var jsonFiles embed.FS

func main() {
    fmt.Println("所有文本文件:")
    fs.WalkDir(textFiles, ".", func(path string, d fs.DirEntry, err error) error {
        if err != nil || d.IsDir() {
            return err
        }
        fmt.Printf("  - %s\n", path)
        return nil
    })
    
    fmt.Println("\n所有 JSON 文件:")
    fs.WalkDir(jsonFiles, ".", func(path string, d fs.DirEntry, err error) error {
        if err != nil || d.IsDir() {
            return err
        }
        fmt.Printf("  - %s\n", path)
        return nil
    })
}

⚠️ Embed 使用注意事项

  • //go:embed 指令位置：必须在变量声明紧上方，中间不能有空行
  • 变量类型限制：只能是 string、[]byte 或 embed.FS 类型
  • 路径限制：路径必须是相对路径，不能包含 .. 或绝对路径
  • 文件大小：嵌入大文件会增加二进制文件大小
  • 跨平台路径：使用 / 作为路径分隔符，自动处理不同系统
  • 构建限制：go run 支持嵌入，但某些构建工具可能不支持

完整的 Web 服务器示例

package main

import (
    "embed"
    "fmt"
    "io/fs"
    "log"
    "net/http"
)

//go:embed static
var staticFS embed.FS

//go:embed templates/*.html
var templatesFS embed.FS

func main() {
    // 设置静态文件服务
    staticFiles, err := fs.Sub(staticFS, "static")
    if err != nil {
        log.Fatal(err)
    }
    http.Handle("/static/", http.FileServer(http.FS(staticFiles)))
    
    // 设置主页
    http.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
        content, err := fs.ReadFile(templatesFS, "templates/index.html")
        if err != nil {
            http.Error(w, err.Error(), http.StatusInternalServerError)
            return
        }
        w.Header().Set("Content-Type", "text/html; charset=utf-8")
        w.Write(content)
    })
    
    // API 端点
    http.HandleFunc("/api/status", func(w http.ResponseWriter, r *http.Request) {
        w.Header().Set("Content-Type", "application/json")
        fmt.Fprint(w, `{"status": "ok", "message": "Server is running"}`)
    })
    
    fmt.Println("服务器启动在 http://localhost:8080")
    log.Fatal(http.ListenAndServe(":8080", nil))
}
Embed 支持的类型和用法
类型 用途 示例 适用场景
string 嵌入为字符串 var s string 文本文件、配置文件
[]byte 嵌入为字节切片 var b []byte 二进制文件、图片、压缩包
embed.FS 嵌入为文件系统 var fsys embed.FS 目录、多个文件

✅ Embed 的最佳实践

  • 小文件优先：只嵌入必要的小文件，避免嵌入大文件
  • 静态资源：适合嵌入静态 HTML、CSS、JS 等资源
  • 配置文件：适合嵌入默认配置或示例配置
  • 版本控制：嵌入的文件会被包含在版本控制中
  • 构建优化：使用构建标签控制不同环境下的嵌入
