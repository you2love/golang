Testify - Go 测试框架

Testify 是 Go 最流行的测试库，提供了丰富的断言函数和测试工具。掌握 Testify 可以编写更清晰、更易维护的测试代码。

断言函数

📝 常用断言

package main

import (
    "testing"
    "github.com/stretchr/testify/assert"
)

func TestAssertions(t *testing.T) {
    // 相等性
    assert.Equal(t, 1, 1)
    assert.NotEqual(t, 1, 2)
    assert.Exactly(t, 1, 1)
    
    // 布尔值
    assert.True(t, true)
    assert.False(t, false)
    
    // 空值检查
    assert.Nil(t, nil)
    assert.NotNil(t, "value")
    assert.Empty(t, []int{})
    assert.NotEmpty(t, []int{1})
    
    // 包含检查
    assert.Contains(t, "hello world", "world")
    assert.NotContains(t, "hello", "world")
    
    // 切片/数组
    assert.ElementsMatch(t, []int{1, 2}, []int{2, 1})
    assert.Subset(t, []int{1, 2, 3}, []int{1, 2})
    
    // 错误检查
    err := someFunc()
    assert.NoError(t, err)
    assert.Error(t, err)
    assert.ErrorIs(t, err, os.ErrNotExist)
    
    // 比较
    assert.Greater(t, 5, 3)
    assert.Less(t, 3, 5)
    assert.InRange(t, 5, 1, 10)
    
    // 自定义消息
    assert.Equal(t, 1, 1, "Values should be equal")
}

func someFunc() error {
    return nil
}

测试套件

📝 Suite 组织测试

package main

import (
    "testing"
    "github.com/stretchr/testify/suite"
)

type ExampleTestSuite struct {
    suite.Suite
    data []int
}

// 每个测试前运行
func (s *ExampleTestSuite) SetupTest() {
    s.data = []int{1, 2, 3}
}

// 每个测试后运行
func (s *ExampleTestSuite) TearDownTest() {
    s.data = nil
}

// 所有测试前运行
func (s *ExampleTestSuite) SetupSuite() {
    // 初始化共享资源
}

// 所有测试后运行
func (s *ExampleTestSuite) TearDownSuite() {
    // 清理共享资源
}

func (s *ExampleTestSuite) TestExample1() {
    s.Len(s.data, 3)
    s.Contains(s.data, 2)
}

func (s *ExampleTestSuite) TestExample2() {
    s.Greater(s.data[2], s.data[0])
}

// 运行套件
func TestExampleTestSuite(t *testing.T) {
    suite.Run(t, &ExampleTestSuite{})
}

Mock 模拟

📝 Mock 对象

package main

import (
    "testing"
    "github.com/stretchr/testify/mock"
)

// 定义 Mock 结构
type MockDB struct {
    mock.Mock
}

func (m *MockDB) GetUser(id int) (*User, error) {
    args := m.Called(id)
    if args.Get(0) == nil {
        return nil, args.Error(1)
    }
    return args.Get(0).(*User), args.Error(1)
}

func TestMockDB(t *testing.T) {
    mockDB := &MockDB{}
    
    // 设置期望
    mockDB.On("GetUser", 1).Return(&User{Name: "Alice"}, nil)
    mockDB.On("GetUser", 999).Return(nil, errors.New("not found"))
    
    // 调用
    user, err := mockDB.GetUser(1)
    
    // 断言
    assert.NotNil(t, user)
    assert.Equal(t, "Alice", user.Name)
    assert.NoError(t, err)
    
    // 验证期望被调用
    mockDB.AssertExpectations(t)
    mockDB.AssertCalled(t, "GetUser", 1)
}

Require vs Assert

📝 Require 必须通过

import "github.com/stretchr/testify/require"

func TestRequire(t *testing.T) {
    // require 失败时会终止测试
    require.NotNil(t, data)  // 失败则停止
    
    // assert 失败后继续执行
    assert.Equal(t, 1, 1)  // 失败也继续
    
    // 使用场景:
    // require: 前置条件检查
    // assert: 结果验证
}
