io 输入输出

io.Reader 接口

io.Reader 是 Go 语言中最基础的读取接口，定义了从数据源读取数据的规范。

type Reader interface {
    Read(p []byte) (n int, err error)
}

💡 Reader 接口说明

  • 参数 p：用于存储读取数据的字节切片
  • 返回值 n：实际读取的字节数（0 ≤ n ≤ len(p)）
  • 返回值 err：错误信息，io.EOF 表示读取结束
  • 调用契约：即使读取到错误，也可能返回部分数据
package main

import (
    "fmt"
    "io"
    "strings"
)

func main() {
    // strings.Reader 实现了 io.Reader 接口
    reader := strings.NewReader("Hello, Go!")
    
    // 创建缓冲区
    buf := make([]byte, 4)
    
    // 循环读取
    for {
        n, err := reader.Read(buf)
        if err != nil {
            if err == io.EOF {
                fmt.Println("读取完成")
            } else {
                fmt.Println("错误:", err)
            }
            break
        }
        fmt.Printf("读取了 %d 字节: %s\n", n, string(buf[:n]))
    }
}

io.Writer 接口

io.Writer 是 Go 语言中最基础的写入接口，定义了向数据目标写入数据的规范。

type Writer interface {
    Write(p []byte) (n int, err error)
}

💡 Writer 接口说明

  • 参数 p：要写入的数据字节切片
  • 返回值 n：实际写入的字节数（0 ≤ n ≤ len(p)）
  • 返回值 err：错误信息，nil 表示成功
  • 调用契约：即使返回错误，也可能写入部分数据
package main

import (
    "fmt"
    "io"
    "os"
)

func main() {
    // os.Stdout 实现了 io.Writer 接口
    writer := os.Stdout
    
    // 写入数据
    data := []byte("Hello, Go!\n")
    n, err := writer.Write(data)
    
    if err != nil {
        fmt.Println("写入错误:", err)
    } else {
        fmt.Printf("成功写入 %d 字节\n", n)
    }
}

接口组合（Interface Composition）

Go 语言通过接口组合来创建更强大的接口。io 包定义了多个组合接口，将 Reader 和 Writer 与其他接口组合。

io 包中的组合接口
接口 组合方式 说明 典型实现
io.ReadWriter Reader + Writer 可读可写 net.Conn, os.File
io.ReadCloser Reader + Closer 可读可关闭 os.File, http.Response.Body
io.WriteCloser Writer + Closer 可写可关闭 os.File, http.Request.Body
io.ReadWriteCloser Reader + Writer + Closer 可读可写可关闭 net.Conn, os.File
io.Seeker 独立接口 可定位读写位置 os.File, bytes.Reader
io.ReadSeeker Reader + Seeker 可读可定位 os.File, bytes.Reader
io.WriteSeeker Writer + Seeker 可写可定位 os.File
io.ReadWriteSeeker Reader + Writer + Seeker 可读可写可定位 os.File

✅ 接口组合的优势

  • 灵活性：可以根据需要组合不同的接口
  • 类型安全：编译时检查接口实现
  • 代码复用：避免重复定义相似的功能
  • 向后兼容：可以轻松扩展现有接口

自定义组合接口

package main

import (
    "fmt"
    "io"
)

// 自定义组合接口
type FileHandler interface {
    io.Reader
    io.Writer
    io.Closer
    Name() string
}

type MyFile struct {
    name string
    data []byte
    pos  int
}

func (f *MyFile) Read(p []byte) (n int, err error) {
    if f.pos >= len(f.data) {
        return 0, io.EOF
    }
    n = copy(p, f.data[f.pos:])
    f.pos += n
    return n, nil
}

func (f *MyFile) Write(p []byte) (n int, err error) {
    f.data = append(f.data, p...)
    return len(p), nil
}

func (f *MyFile) Close() error {
    fmt.Printf("File %s closed\n", f.name)
    return nil
}

func (f *MyFile) Name() string {
    return f.name
}

func processFile(handler FileHandler) error {
    fmt.Printf("Processing file: %s\n", handler.Name())
    
    // 写入数据
    _, err := handler.Write([]byte("Hello, World!"))
    if err != nil {
        return err
    }
    
    // 读取数据
    buf := make([]byte, 100)
    n, err := handler.Read(buf)
    if err != nil && err != io.EOF {
        return err
    }
    fmt.Printf("Read: %s\n", string(buf[:n]))
    
    // 关闭文件
    return handler.Close()
}

func main() {
    file := &MyFile{
        name: "test.txt",
        data: []byte("Initial content"),
    }
    
    err := processFile(file)
    if err != nil {
        fmt.Println("Error:", err)
    }
}

接口组合的实际应用

package main

import (
    "bytes"
    "fmt"
    "io"
)

// 通用的数据处理器
func processData(r io.Reader, w io.Writer) error {
    // 从 Reader 读取数据
    data, err := io.ReadAll(r)
    if err != nil {
        return err
    }
    
    // 处理数据（转换为大写）
    for i := range data {
        if data[i] >= 'a' && data[i] <= 'z' {
            data[i] -= 32
        }
    }
    
    // 写入到 Writer
    _, err = w.Write(data)
    return err
}

func main() {
    // 使用 bytes.Buffer 作为 Reader 和 Writer
    input := bytes.NewBufferString("hello, world!")
    output := &bytes.Buffer{}
    
    // 处理数据
    err := processData(input, output)
    if err != nil {
        fmt.Println("Error:", err)
        return
    }
    
    fmt.Println("Input:", input.String())
    fmt.Println("Output:", output.String())
}

⚠️ 接口使用注意事项

  • 接口是隐式实现的：不需要显式声明实现某个接口
  • 接口值是 nil 安全的：调用 nil 接口的方法会 panic
  • 接口组合是类型安全的：编译时会检查所有方法是否实现
  • 避免过度组合：接口应该小而专注，遵循最小接口原则
  • 文档化接口契约：清晰说明接口的行为和约束

读取文件

package main

import (
    "fmt"
    "io"
    "os"
)

func main() {
    // 打开文件
    file, err := os.Open("test.txt")
    if err != nil {
        fmt.Println("Error:", err)
        return
    }
    defer file.Close()
    
    // 读取整个文件
    data, err := io.ReadAll(file)
    if err != nil {
        fmt.Println("Error:", err)
        return
    }
    fmt.Println("File content:", string(data))
}

写入文件

package main

import (
    "fmt"
    "os"
)

func main() {
    // 写入文件（覆盖）
    err := os.WriteFile("output.txt", []byte("Hello, World!"), 0644)
    if err != nil {
        fmt.Println("Error:", err)
        return
    }
    
    // 追加写入
    f, err := os.OpenFile("output.txt", os.O_APPEND|os.O_WRONLY, 0644)
    if err != nil {
        fmt.Println("Error:", err)
        return
    }
    defer f.Close()
    
    _, err = f.WriteString("\nAppended text")
    if err != nil {
        fmt.Println("Error:", err)
    }
}

创建和删除文件/目录

package main

import (
    "fmt"
    "os"
)

func main() {
    // 创建目录
    err := os.MkdirAll("testdir/subdir", 0755)
    if err != nil {
        fmt.Println("Error:", err)
    }
    
    // 创建文件
    file, err := os.Create("testdir/test.txt")
    if err != nil {
        fmt.Println("Error:", err)
    }
    file.Close()
    
    // 删除文件
    os.Remove("testdir/test.txt")
    
    // 删除目录
    os.RemoveAll("testdir")
}

使用 io.Copy

package main

import (
    "io"
    "os"
)

func copyFile(src, dst string) error {
    source, err := os.Open(src)
    if err != nil {
        return err
    }
    defer source.Close()
    
    destination, err := os.Create(dst)
    if err != nil {
        return err
    }
    defer destination.Close()
    
    _, err = io.Copy(destination, source)
    return err
}

使用 io.LimitReader

package main

import (
    "fmt"
    "io"
    "strings"
)

func main() {
    reader := strings.NewReader("Hello, World! This is a long string.")
    
    // 限制读取 5 个字节
    limitedReader := io.LimitReader(reader, 5)
    
    data, _ := io.ReadAll(limitedReader)
    fmt.Println("Limited read:", string(data))
}
