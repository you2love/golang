Analysis 框架 - golang.org/x/tools/analysis

golang.org/x/tools/analysis 是 Go 官方提供的代码分析框架，用于构建静态分析工具。它提供了统一的接口和基础设施，简化了分析器的开发。

    Doc:  "示例分析器",
    Run:  run,

    // 遍历所有文件
    for _, file := range pass.Files {
        // 分析 AST
        // ...
    }
    return nil, nil
}

func main() {
    singlechecker.Main(Analyzer)
}

诊断报告

📝 报告问题

package main

import (
    "fmt"
    "go/ast"
    "golang.org/x/tools/go/analysis"
    "golang.org/x/tools/go/analysis/singlechecker"
)

var Analyzer = &analysis.Analyzer{
    Name: "unreachable",
    Doc:  "检查 unreachable 代码",
    Run:  run,
}

func run(pass *analysis.Pass) (interface{}, error) {
    for _, file := range pass.Files {
        ast.Inspect(file, func(n ast.Node) bool {
            // 检查 return 后的代码
            if stmt, ok := n.(*ast.ReturnStmt); ok {
                // 报告问题
                pass.Reportf(stmt.Pos(), "unreachable code after return")
            }
            return true
        })
    }
    return nil, nil
}

func main() {
    singlechecker.Main(Analyzer)
}

Reportf 方法

// Reportf 报告诊断信息

    Message: "variable is unused",

            Message: "Remove unused variable",
            TextEdits: []analysis.TextEdit{
                {
                    Pos:     start,
                    End:     end,
                    NewText: []byte(""),
                },
            },
        },
    },
})

遍历 AST

📝 使用 Inspector

package main

import (
    "go/ast"
    "golang.org/x/tools/go/analysis"
    "golang.org/x/tools/go/analysis/passes/inspect"
    "golang.org/x/tools/go/ast/inspector"
)

var Analyzer = &analysis.Analyzer{
    Name:     "myanalyzer",
    Doc:      "使用 Inspector 遍历 AST",
    Run:      run,
    Requires: []*analysis.Analyzer{inspect.Analyzer},
}

func run(pass *analysis.Pass) (interface{}, error) {
    // 获取 Inspector
    inspect := pass.ResultOf[inspect.Analyzer].(*inspector.Inspector)

    // 指定要检查的节点类型
    nodeFilter := []ast.Node{
        (*ast.FuncDecl)(nil),
        (*ast.CallExpr)(nil),
    }

    // 遍历节点
    inspect.Preorder(nodeFilter, func(n ast.Node) {
        switch n := n.(type) {
        case *ast.FuncDecl:
            // 处理函数声明
            pass.Reportf(n.Pos(), "found function: %s", n.Name.Name)
        case *ast.CallExpr:
            // 处理函数调用
            if ident, ok := n.Fun.(*ast.Ident); ok {
                pass.Reportf(n.Pos(), "found call: %s", ident.Name)
            }
        }
    })

    return nil, nil
}

Facts - 跨包分析

📝 导出和导入 Facts

package main

import (
    "fmt"
    "go/ast"
    "golang.org/x/tools/go/analysis"
)

// 定义 Fact 类型
type MyFact struct {
    Data string
}

func (f *MyFact) AFact()         {}
func (f *MyFact) String() string { return f.Data }

var Analyzer = &analysis.Analyzer{
    Name: "myfact",
    Doc:  "导出和导入 facts",
    Run:  run,
    // 声明 Fact 类型
    FactTypes: []analysis.Fact{&MyFact{}},
}

func run(pass *analysis.Pass) (interface{}, error) {
    // 导出 Fact
    for _, file := range pass.Files {
        ast.Inspect(file, func(n ast.Node) bool {
            if fn, ok := n.(*ast.FuncDecl); ok {
                // 导出关于函数的 fact
                pass.ExportObjectFact(fn.Obj(), &MyFact{Data: "exported"})
            }
            return true
        })
    }

    // 导入 Fact
    var fact MyFact
    if pass.ImportObjectFact(someObj, &fact) {
        fmt.Printf("Found fact: %s\n", fact.Data)
    }

    return nil, nil
}

实际示例

检查未使用的变量

package main

import (
    "go/ast"
    "golang.org/x/tools/go/analysis"
    "golang.org/x/tools/go/analysis/singlechecker"
)

var Analyzer = &analysis.Analyzer{
    Name: "unusedvar",
    Doc:  "检查未使用的变量",
    Run:  run,
}

func run(pass *analysis.Pass) (interface{}, error) {
    for _, file := range pass.Files {
        // 收集所有声明的变量
        declared := make(map[string]bool)
        ast.Inspect(file, func(n ast.Node) bool {
            if vs, ok := n.(*ast.ValueSpec); ok {
                for _, name := range vs.Names {
                    declared[name.Name] = true
                }
            }
            return true
        })

        // 收集所有使用的变量
        used := make(map[string]bool)
        ast.Inspect(file, func(n ast.Node) bool {
            if ident, ok := n.(*ast.Ident); ok {
                used[ident.Name] = true
            }
            return true
        })

        // 报告未使用的变量
        for name := range declared {
            if !used[name] {
                pass.Reportf(token.NoPos, "variable %s is unused", name)
            }
        }
    }
    return nil, nil
}

func main() {
    singlechecker.Main(Analyzer)
}

检查函数复杂度

package main

import (
    "go/ast"
    "golang.org/x/tools/go/analysis"
    "golang.org/x/tools/go/analysis/singlechecker"
)

var Analyzer = &analysis.Analyzer{
    Name: "funccomplexity",
    Doc:  "检查函数复杂度",
    Run:  run,
}

func run(pass *analysis.Pass) (interface{}, error) {
    for _, file := range pass.Files {
        ast.Inspect(file, func(n ast.Node) bool {
            if fn, ok := n.(*ast.FuncDecl); ok {
                complexity := calculateComplexity(fn.Body)
                if complexity > 10 {
                    pass.Reportf(fn.Pos(), "function %s has complexity %d",
                        fn.Name.Name, complexity)
                }
            }
            return true
        })
    }
    return nil, nil
}

func calculateComplexity(body *ast.BlockStmt) int {
    complexity := 1
    ast.Inspect(body, func(n ast.Node) bool {
        switch n.(type) {
        case *ast.IfStmt, *ast.ForStmt, *ast.RangeStmt,
             *ast.CaseClause, *ast.CommClause:
            complexity++
        }
        return true
    })
    return complexity
}

func main() {
    singlechecker.Main(Analyzer)
}

最佳实践

✅ 使用建议

  • 单一职责: 每个 Analyzer 只做一件事
  • 清晰文档: 提供清晰的 Doc 说明
  • 准确报告: 使用准确的位置和消息
  • 提供修复: 尽可能提供 SuggestedFixes
  • 性能考虑: 避免重复遍历 AST
  • 使用 Inspector: 优先使用 Inspector 而非 Inspect
