语法树 - go/ast

go/ast 包提供了 Go 抽象语法树 (AST) 的数据结构和操作函数。AST 是编译器对源代码的结构化表示，用于代码分析、转换和生成。

基础用法

解析文件

📝 解析 Go 文件

package main

import (
    "fmt"
    "go/ast"
    "go/parser"
    "go/token"
)

func main() {
    // 源代码
    src := `package main

import "fmt"

func main() {
    fmt.Println("Hello")
}`

    // 创建文件集
    fset := token.NewFileSet()

    // 解析源代码
    file, err := parser.ParseFile(fset, "", src, 0)
    if err != nil {
        panic(err)
    }

    // 打印 AST
    ast.Fprint(nil, fset, file, nil)
}

遍历 AST

使用 Inspect

📝 遍历所有节点

package main

import (
    "fmt"
    "go/ast"
    "go/parser"
    "go/token"
)

func main() {
    src := `package main

import "fmt"

func main() {
    fmt.Println("Hello")
}`

    fset := token.NewFileSet()
    file, _ := parser.ParseFile(fset, "", src, 0)

    // 使用 Inspect 遍历
    ast.Inspect(file, func(n ast.Node) bool {
        if n != nil {
            fmt.Printf("%T\n", n)
        }
        return true // 继续遍历子节点
    })
}

查找特定节点

📝 查找函数声明

package main

import (
    "fmt"
    "go/ast"
    "go/parser"
    "go/token"
)

func main() {
    src := `package main

func add(a, b int) int {
    return a + b
}

func main() {
    result := add(1, 2)
    fmt.Println(result)
}`

    fset := token.NewFileSet()
    file, _ := parser.ParseFile(fset, "", src, 0)

    // 查找所有函数
    ast.Inspect(file, func(n ast.Node) bool {
        if fn, ok := n.(*ast.FuncDecl); ok {
            fmt.Printf("Function: %s\n", fn.Name.Name)
            
            // 打印参数
            if fn.Type.Params != nil {
                fmt.Print("  Parameters: ")
                for i, param := range fn.Type.Params.List {
                    if i > 0 {
                        fmt.Print(", ")
                    }
                    for j, name := range param.Names {
                        if j > 0 {
                            fmt.Print(", ")
                        }
                        fmt.Print(name.Name)
                    }
                }
                fmt.Println()
            }
            
            // 打印返回值
            if fn.Type.Results != nil && len(fn.Type.Results.List) > 0 {
                resultType := fn.Type.Results.List[0].Type
                fmt.Printf("  Returns: %T\n", resultType)
            }
        }
        return true
    })
}

实用示例

统计函数数量

📝 代码统计工具

package main

import (
    "fmt"
    "go/ast"
    "go/parser"
    "go/token"
)

type CodeStats struct {
    Functions   int
    Methods     int
    Imports     int
    Structs     int
    Interfaces  int
}

func AnalyzeFile(filename string) (*CodeStats, error) {
    fset := token.NewFileSet()
    file, err := parser.ParseFile(fset, filename, nil, 0)
    if err != nil {
        return nil, err
    }
    
    stats := &CodeStats{}
    
    ast.Inspect(file, func(n ast.Node) bool {
        switch x := n.(type) {
        case *ast.FuncDecl:
            if x.Recv != nil {
                stats.Methods++
            } else {
                stats.Functions++
            }
        case *ast.GenDecl:
            if x.Tok == token.IMPORT {
                stats.Imports += len(x.Specs)
            }
            for _, spec := range x.Specs {
                switch spec.(type) {
                case *ast.TypeSpec:
                    ts := spec.(*ast.TypeSpec)
                    switch ts.Type.(type) {
                    case *ast.StructType:
                        stats.Structs++
                    case *ast.InterfaceType:
                        stats.Interfaces++
                    }
                }
            }
        }
        return true
    })
    
    return stats, nil
}

func main() {
    stats, err := AnalyzeFile("main.go")
    if err != nil {
        panic(err)
    }
    
    fmt.Printf("Functions: %d\n", stats.Functions)
    fmt.Printf("Methods: %d\n", stats.Methods)
    fmt.Printf("Imports: %d\n", stats.Imports)
    fmt.Printf("Structs: %d\n", stats.Structs)
    fmt.Printf("Interfaces: %d\n", stats.Interfaces)
}

查找所有调用

📝 函数调用分析

package main

import (
    "fmt"
    "go/ast"
    "go/parser"
    "go/token"
)

func FindCallExprs(filename string) {
    fset := token.NewFileSet()
    file, _ := parser.ParseFile(fset, filename, nil, 0)
    
    ast.Inspect(file, func(n ast.Node) bool {
        if call, ok := n.(*ast.CallExpr); ok {
            // 获取被调用的函数名
            switch fun := call.Fun.(type) {
            case *ast.Ident:
                fmt.Printf("Call: %s()\n", fun.Name)
            case *ast.SelectorExpr:
                if ident, ok := fun.X.(*ast.Ident); ok {
                    fmt.Printf("Call: %s.%s()\n",
                        ident.Name, fun.Sel.Name)
                }
            }
            
            // 打印参数数量
            fmt.Printf("  Arguments: %d\n", len(call.Args))
        }
        return true
    })
}

func main() {
    FindCallExprs("main.go")
}

生成代码

📝 AST 代码生成

package main

import (
    "bytes"
    "fmt"
    "go/ast"
    "go/format"
    "go/token"
)

func GenerateHello() []byte {
    // 创建文件
    file := &ast.File{
        Name: ast.NewIdent("main"),
    }
    
    // 创建 main 函数
    mainFunc := &ast.FuncDecl{
        Name: ast.NewIdent("main"),
        Type: &ast.FuncType{
            Params: &ast.FieldList{},
        },
        Body: &ast.BlockStmt{
            List: []ast.Stmt{
                &ast.ExprStmt{
                    X: &ast.CallExpr{
                        Fun: &ast.SelectorExpr{
                            X:  ast.NewIdent("fmt"),
                            Sel: ast.NewIdent("Println"),
                        },
                        Args: []ast.Expr{
                            &ast.BasicLit{
                                Kind:  token.STRING,
                                Value: `"Hello, World!"`,
                            },
                        },
                    },
                },
            },
        },
    }
    
    file.Decls = []ast.Decl{mainFunc}
    
    // 格式化输出
    var buf bytes.Buffer
    fset := token.NewFileSet()
    format.Node(&buf, fset, file)
    
    return buf.Bytes()
}

func main() {
    code := GenerateHello()
    fmt.Print(string(code))
}

常见节点类型

📖 节点类型对照

// 声明节点
*ast.FuncDecl      // 函数声明
*ast.GenDecl       // 通用声明 (import, var, const, type)
*ast.ImportSpec    // import 声明
*ast.ValueSpec     // var/const 声明
*ast.TypeSpec      // type 声明

// 语句节点
*ast.BlockStmt     // 代码块 { }
*ast.IfStmt        // if 语句
*ast.ForStmt       // for 循环
*ast.ReturnStmt    // return 语句
*ast.AssignStmt    // 赋值语句
*ast.ExprStmt      // 表达式语句

// 表达式节点
*ast.Ident         // 标识符
*ast.BasicLit      // 字面量
*ast.CallExpr      // 函数调用
*ast.SelectorExpr  // 选择器 (x.sel)
*ast.BinaryExpr    // 二元表达式
*ast.UnaryExpr     // 一元表达式

// 类型节点
*ast.StructType    // 结构体类型
*ast.InterfaceType // 接口类型
*ast.ArrayType     // 数组类型
*ast.MapType       // Map 类型
*ast.ChanType      // Channel 类型

最佳实践

✅ 使用建议

  • 使用 FileSet: 始终使用 token.FileSet 管理位置
  • 类型断言: 使用 type switch 处理不同节点
  • Inspect 返回: 返回 false 可跳过子节点遍历
  • 格式化输出: 使用 format.Node 格式化 AST
  • 错误处理: 解析时始终检查错误
