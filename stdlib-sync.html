sync 并发原语

Mutex 互斥锁

package main

import (
    "fmt"
    "sync"
)

type Counter struct {
    mu    sync.Mutex
    value int
}

func (c *Counter) Increment() {
    c.mu.Lock()
    defer c.mu.Unlock()
    c.value++
}

func (c *Counter) Value() int {
    c.mu.Lock()
    defer c.mu.Unlock()
    return c.value
}

func main() {
    var counter Counter
    var wg sync.WaitGroup
    
    for i := 0; i < 1000; i++ {
        wg.Add(1)
        go func() {
            defer wg.Done()
            counter.Increment()
        }()
    }
    
    wg.Wait()
    fmt.Println("Final value:", counter.Value())
}

RWMutex 读写锁

package main

import (
    "fmt"
    "sync"
    "time"
)

type Cache struct {
    mu    sync.RWMutex
    data  map[string]string
}

func NewCache() *Cache {
    return &Cache{
        data: make(map[string]string),
    }
}

func (c *Cache) Get(key string) (string, bool) {
    c.mu.RLock()
    defer c.mu.RUnlock()
    val, ok := c.data[key]
    return val, ok
}

func (c *Cache) Set(key, value string) {
    c.mu.Lock()
    defer c.mu.Unlock()
    c.data[key] = value
}

func main() {
    cache := NewCache()
    var wg sync.WaitGroup
    
    // 写入
    for i := 0; i < 10; i++ {
        wg.Add(1)
        go func(i int) {
            defer wg.Done()
            key := fmt.Sprintf("key%d", i)
            value := fmt.Sprintf("value%d", i)
            cache.Set(key, value)
        }(i)
    }
    
    // 读取
    for i := 0; i < 10; i++ {
        wg.Add(1)
        go func(i int) {
            defer wg.Done()
            key := fmt.Sprintf("key%d", i)
            time.Sleep(time.Millisecond)
            val, ok := cache.Get(key)
            fmt.Printf("Get %s: %v, %v\n", key, val, ok)
        }(i)
    }
    
    wg.Wait()
}

WaitGroup 等待组

package main

import (
    "fmt"
    "sync"
)

func worker(id int, wg *sync.WaitGroup) {
    defer wg.Done()
    fmt.Printf("Worker %d started\n", id)
    fmt.Printf("Worker %d finished\n", id)
}

func main() {
    var wg sync.WaitGroup
    
    for i := 1; i <= 5; i++ {
        wg.Add(1)
        go worker(i, &wg)
    }
    
    wg.Wait()
    fmt.Println("All workers finished")
}

Once 单次执行

package main

import (
    "fmt"
    "sync"
)

var (
    once     sync.Once
    instance *Singleton
)

type Singleton struct {
    value string
}

func GetInstance() *Singleton {
    once.Do(func() {
        instance = &Singleton{value: "initialized"}
        fmt.Println("Singleton initialized")
    })
    return instance
}

func main() {
    // 多次调用只会初始化一次
    GetInstance()
    GetInstance()
    GetInstance()
}

Cond 条件变量

package main

import (
    "fmt"
    "sync"
    "time"
)

func main() {
    var mu sync.Mutex
    cond := sync.NewCond(&mu)
    ready := false
    
    // 等待者
    go func() {
        mu.Lock()
        for !ready {
            fmt.Println("Waiting...")
            cond.Wait()
        }
        fmt.Println("Ready!")
        mu.Unlock()
    }()
    
    time.Sleep(time.Second)
    
    // 通知者
    mu.Lock()
    ready = true
    fmt.Println("Signaling...")
    cond.Signal()
    mu.Unlock()
    
    time.Sleep(time.Second)
}

Pool 对象池

package main

import (
    "bytes"
    "fmt"
    "sync"
)

var bufferPool = sync.Pool{
    New: func() interface{} {
        fmt.Println("Creating new buffer")
        return &bytes.Buffer{}
    },
}

func process() {
    // 从池中获取
    buf := bufferPool.Get().(*bytes.Buffer)
    defer func() {
        // 重置并放回池中
        buf.Reset()
        bufferPool.Put(buf)
    }()
    
    buf.WriteString("Hello, World!")
    fmt.Println("Buffer:", buf.String())
}

func main() {
    for i := 0; i < 5; i++ {
        process()
    }
}

Map 并发安全映射

package main

import (
    "fmt"
    "sync"
)

func main() {
    var m sync.Map
    
    // 存储
    m.Store("key1", "value1")
    m.Store("key2", "value2")
    
    // 加载
    if val, ok := m.Load("key1"); ok {
        fmt.Println("key1:", val)
    }
    
    // 加载或存储
    actual, loaded := m.LoadOrStore("key3", "value3")
    fmt.Printf("key3: %v, loaded: %v\n", actual, loaded)
    
    // 遍历
    m.Range(func(key, value interface{}) bool {
        fmt.Printf("%s: %s\n", key, value)
        return true
    })
    
    // 删除
    m.Delete("key1")
}

Atomic 原子操作

package main

import (
    "fmt"
    "sync/atomic"
)

func main() {
    var counter int64
    
    // 原子增加
    atomic.AddInt64(&counter, 1)
    fmt.Println("Counter:", counter)
    
    // 原子加载
    value := atomic.LoadInt64(&counter)
    fmt.Println("Loaded:", value)
    
    // 原子存储
    atomic.StoreInt64(&counter, 100)
    fmt.Println("Stored:", atomic.LoadInt64(&counter))
    
    // 原子比较并交换
    swapped := atomic.CompareAndSwapInt64(&counter, 100, 200)
    fmt.Printf("Swapped: %v, Value: %d\n", swapped, atomic.LoadInt64(&counter))
}

noCopy 禁止复制

noCopy 是 sync 包中的一个重要机制，用于防止并发类型被意外复制。

⚠️ 为什么需要 noCopy？

  • 状态一致性：sync 包中的类型（如 Mutex、WaitGroup）包含内部状态，复制会导致状态不一致
  • 死锁风险：复制的锁可能导致死锁或不可预期的行为
  • 内存安全：复制可能导致内存泄漏或竞争条件
  • API 契约：sync 类型应该通过指针传递，而不是值传递
// noCopy 的定义（在 sync 包中）
type noCopy struct{}

// Mutex 的定义（嵌入了 noCopy）
type Mutex struct {
    state int32
    sema  uint32
    noCopy noCopy  // 嵌入 noCopy，防止复制
}

错误的复制示例

package main

import (
    "fmt"
    "sync"
)

type SafeCounter struct {
    mu    sync.Mutex
    value int
}

// ❌ 错误：通过值复制了 Mutex
func badCopyExample() {
    counter := SafeCounter{
        value: 0,
    }
    
    // 错误：复制了整个结构体（包括 Mutex）
    counterCopy := counter  // 这会复制 Mutex！
    
    counterCopy.mu.Lock()
    counterCopy.value++
    counterCopy.mu.Unlock()
    
    // 原始 counter 的状态是独立的
    fmt.Printf("Original: %d, Copy: %d\n", counter.value, counterCopy.value)
}

// ✅ 正确：使用指针传递
func correctPointerExample() {
    counter := &SafeCounter{
        value: 0,
    }
    
    // 正确：传递指针
    incrementCounter(counter)
}

func incrementCounter(c *SafeCounter) {
    c.mu.Lock()
    defer c.mu.Unlock()
    c.value++
}

func main() {
    fmt.Println("运行 badCopyExample...")
    badCopyExample()
    
    fmt.Println("\n运行 correctPointerExample...")
    correctPointerExample()
}

使用 go vet 检测复制

package main

import (
    "fmt"
    "sync"
)

type Resource struct {
    mu sync.Mutex
    data map[string]int
}

// ❌ 错误：函数参数使用值传递
// go vet 会检测到这个问题
func processResource(r Resource) {  // 错误：复制了 Mutex
    r.mu.Lock()
    defer r.mu.Unlock()
    r.data["key"] = 42
}

// ✅ 正确：函数参数使用指针传递
func processResourceSafe(r *Resource) {  // 正确：使用指针
    r.mu.Lock()
    defer r.mu.Unlock()
    r.data["key"] = 42
}

func main() {
    resource := &Resource{
        data: make(map[string]int),
    }
    
    // 使用 go vet 检测复制问题
    // 运行命令：go vet main.go
    
    fmt.Println("使用 go vet 检测复制问题...")
    fmt.Println("运行 'go vet' 命令会检测到 processResource 函数的复制问题")
}

💡 noCopy 工作原理

  • 空结构体：noCopy 是一个空结构体，不占用内存
  • 嵌入到类型中：sync 包中的类型都嵌入了 noCopy
  • go vet 检测：go vet 工具会检查包含 noCopy 的类型是否被复制
  • 编译时警告：虽然不是编译错误，但会发出警告
  • 最佳实践：始终使用指针传递 sync 包中的类型

自定义类型的 noCopy 使用

package main

import (
    "fmt"
    "sync"
)

// 自定义类型嵌入 noCopy
type SafeMap struct {
    mu    sync.Mutex
    data  map[string]interface{}
    noCopy sync.noCopy  // 嵌入 noCopy，防止复制
}

func NewSafeMap() *SafeMap {
    return &SafeMap{
        data: make(map[string]interface{}),
    }
}

func (m *SafeMap) Get(key string) (interface{}, bool) {
    m.mu.Lock()
    defer m.mu.Unlock()
    val, ok := m.data[key]
    return val, ok
}

func (m *SafeMap) Set(key string, value interface{}) {
    m.mu.Lock()
    defer m.mu.Unlock()
    m.data[key] = value
}

func main() {
    sm := NewSafeMap()
    
    sm.Set("name", "Alice")
    sm.Set("age", 30)
    
    if name, ok := sm.Get("name"); ok {
        fmt.Println("Name:", name)
    }
}

WaitGroup 等待组 ❌ 不可复制
Once 单次执行 ❌ 不可复制
Cond 条件变量 ❌ 不可复制
Pool 对象池 ❌ 不可复制
Map 并发安全映射 ❌ 不可复制

⚠️ 使用 noCopy 的最佳实践

  • 始终使用指针：sync 类型的变量应该始终使用指针
  • 函数参数：函数参数应该使用指针类型（*sync.Mutex）
  • 结构体字段：如果结构体包含 sync 类型，也应该使用指针
  • go vet 检查：定期运行 go vet 检测复制问题
  • CI/CD 集成：在 CI/CD 流程中集成 go vet 检查
