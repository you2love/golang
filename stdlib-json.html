json 数据编码

基本编码和解码

package main

import (
    "encoding/json"
    "fmt"
)

type Person struct {
    Name  string `json:"name"`
    Age   int    `json:"age"`
    Email string `json:"email,omitempty"`
}

func main() {
    // 编码
    person := Person{
        Name: "Alice",
        Age:  30,
    }
    
    jsonData, err := json.Marshal(person)
    if err != nil {
        fmt.Println("Error:", err)
        return
    }
    fmt.Println("JSON:", string(jsonData))
    
    // 解码
    var decoded Person
    err = json.Unmarshal(jsonData, &decoded)
    if err != nil {
        fmt.Println("Error:", err)
        return
    }
    fmt.Printf("Decoded: %+v\n", decoded)
}

处理嵌套结构

package main

import (
    "encoding/json"
    "fmt"
)

type Address struct {
    City    string `json:"city"`
    Country string `json:"country"`
}

type User struct {
    Name    string  `json:"name"`
    Age     int     `json:"age"`
    Address Address  `json:"address"`
    Tags    []string `json:"tags"`
}

func main() {
    jsonData := []byte(`{
        "name": "Bob",
        "age": 25,
        "address": {
            "city": "Beijing",
            "country": "China"
        },
        "tags": ["developer", "golang"]
    }`)
    
    var user User
    err := json.Unmarshal(jsonData, &user)
    if err != nil {
        fmt.Println("Error:", err)
        return
    }
    
    fmt.Printf("User: %+v\n", user)
    fmt.Printf("Address: %+v\n", user.Address)
    fmt.Printf("Tags: %v\n", user.Tags)
}

encoding/json/v2

Go 1.24 引入了 encoding/json/v2，这是一个全新的 JSON 实现，提供了更好的性能、更严格的类型安全和更清晰的 API。

💡 json/v2 的主要改进

  • 更好的性能：重新设计的编解码器，性能提升 20-30%
  • 更严格的类型安全：拒绝不安全的类型转换
  • 更清晰的 API：简化的接口，减少混淆
  • 更好的错误信息：提供更详细的错误上下文
  • 支持 JSON Schema：内置 JSON Schema 验证支持

使用 Decoder 和 Encoder

package main

import (
    "encoding/json/v2"
    "fmt"
    "os"
)

type Person struct {
    Name string `json:"name"`
    Age  int    `json:"age"`
}

func main() {
    // 写入 JSON 文件
    file, err := os.Create("people.json")
    if err != nil {
        fmt.Println("Error:", err)
        return
    }
    defer file.Close()
    
    encoder := json.NewEncoder(file)
    encoder.SetIndent("", "  ")
    
    people := []Person{
        {Name: "Alice", Age: 30},
        {Name: "Bob", Age: 25},
    }
    
    err = encoder.Encode(people)
    if err != nil {
        fmt.Println("Error:", err)
        return
    }
    
    // 读取 JSON 文件
    file, err = os.Open("people.json")
    if err != nil {
        fmt.Println("Error:", err)
        return
    }
    defer file.Close()
    
    var result []Person
    decoder := json.NewDecoder(file)
    err = decoder.Decode(&result)
    if err != nil {
        fmt.Println("Error:", err)
        return
    }
    
    fmt.Printf("People: %+v\n", result)
}

严格模式

package main

import (
    "encoding/json/v2"
    "fmt"
)

type Product struct {
    Name     string  `json:"name"`
    Price    float64 `json:"price"`
    Quantity int     json:"quantity"`
}

func main() {
    // 使用严格模式的解码器
    opts := &json.DecodeOptions{
        RejectUnknownFields: true,
        DisallowDuplicateFields: true,
    }
    
    jsonData := []byte(`{
        "name": "Laptop",
        "price": 999.99,
        "quantity": 10,
        "extra_field": "will cause error"
    }`)
    
    var product Product
    decoder := json.NewDecoder(jsonData)
    decoder.SetDecodeOptions(opts)
    
    err := decoder.Decode(&product)
    if err != nil {
        fmt.Println("Strict mode error (expected):", err)
    }
}

自定义类型转换

package main

import (
    "encoding/json/v2"
    "fmt"
    "time"
)

type Event struct {
    Name      string    `json:"name"`
    Timestamp string    `json:"timestamp"`
    Date      time.Time `json:"date,omitempty"`
}

func main() {
    jsonData := []byte(`{
        "name": "Meeting",
        "timestamp": "2024-01-15T10:30:00Z"
    }`)
    
    var event Event
    err := json.Unmarshal(jsonData, &event)
    if err != nil {
        fmt.Println("Error:", err)
        return
    }
    
    // 解析时间字符串
    parsedTime, err := time.Parse(time.RFC3339, event.Timestamp)
    if err != nil {
        fmt.Println("Error parsing time:", err)
        return
    }
    event.Date = parsedTime
    
    fmt.Printf("Event: %+v\n", event)
    fmt.Printf("Date: %s\n", event.Date.Format(time.RFC3339))
}

⚠️ json/v2 vs json

  • 向后兼容：json/v2 与 json/v1 不完全兼容，需要逐步迁移
  • 行为差异：某些边缘情况的处理方式有所不同
  • 性能提升：json/v2 在大多数场景下性能更好
  • 类型安全：json/v2 对类型检查更严格
  • 迁移建议：新项目优先使用 json/v2，旧项目谨慎评估后再迁移
